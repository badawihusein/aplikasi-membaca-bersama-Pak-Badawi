  initAudio() {

  playTone(freq, duration, type='sine', vol=0.15) {
    this.initAudio();
    const osc = this.audioCtx.createOscillator();
    const gain = this.audioCtx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.value = vol;
    gain.gain.exponentialRampToValueAtTime(0.001, this.audioCtx.currentTime + duration);
    osc.connect(gain);
    gain.connect(this.audioCtx.destination);
    osc.start();
    osc.stop(this.audioCtx.currentTime + duration);
  },

  // Background music - simple cheerful loop
  startMusic() {
    this.initAudio();
    if (this.musicOsc) return;
    // Buat melody sederhana
    const melody = [262,294,330,349,392,349,330,294,262,330,392,523,392,330,294,262];
    const noteLen = 0.3;
    let time = this.audioCtx.currentTime;
    this.musicGain = this.audioCtx.createGain();
    this.musicGain.gain.value = 0.06;
    this.musicGain.connect(this.audioCtx.destination);
    
    // Loop melody
    const scheduleLoop = (startTime) => {
      melody.forEach((freq, i) => {
        const osc = this.audioCtx.createOscillator();
        osc.type = 'triangle';
        osc.frequency.value = freq;
        osc.connect(this.musicGain);
        osc.start(startTime + i * noteLen);
        osc.stop(startTime + i * noteLen + noteLen);
      });
      return startTime + melody.length * noteLen;
    };
    
    let nextTime = scheduleLoop(time);
    this.musicInterval = setInterval(() => {
      if (this.audioCtx.currentTime > nextTime - 2) {
        nextTime = scheduleLoop(nextTime);
      }
    }, 1000);
    
    this.musicOn = true;
  },

  stopMusic() {
    if (this.musicInterval) clearInterval(this.musicInterval);
    this.musicOn = false;
    this.musicInterval = null;
    // Tidak bisa stop oscillator yang sudah scheduled, tapi gain diset 0
    if (this.musicGain) this.musicGain.gain.value = 0;
    this.musicGain = null;
  },

  toggleMusic() {
    if (this.musicOn) { this.stopMusic(); }
    else { this.startMusic(); }
    this.updateMusicBtn();
  },

    }
  },

  // === SPEECH SYNTHESIS ===
  speak(text, lang='id-ID') {
    if (!window.speechSynthesis) return;
    window.speechSynthesis.cancel();
    const u = new SpeechSynthesisUtterance(text);
    u.lang = lang;
    u.rate = 0.85;
    u.pitch = 1.2;
    // Coba cari voice Indonesia
    const voices = speechSynthesis.getVoices();
    const idVoice = voices.find(v => v.lang.startsWith('id'));
    if (idVoice) u.voice = idVoice;
    speechSynthesis.speak(u);
  },

  // === SPEECH RECOGNITION ===
  initRecognition() {
    const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
    if (!SR) return false;
    this.recognition = new SR();
    this.recognition.lang = 'id-ID';
    this.recognition.continuous = false;
    this.recognition.interimResults = false;
    return true;
  },

  startRecognition(target, callback) {
    if (!this.recognition && !this.initRecognition()) {
      this.showToast('Speech Recognition tidak tersedia di browser ini');
      return;
    }
    this.recognition.onresult = (e) => {
      const result = e.results[0][0].transcript.toLowerCase().trim();
      const isCorrect = result === target.toLowerCase();
      callback(isCorrect, result);
    };
    this.recognition.onerror = (e) => {
      callback(false, 'Error: ' + e.error);
    };
    this.recognition.onend = () => {
      const micBtn = document.querySelector('.mic-btn');
      if (micBtn) micBtn.classList.remove('recording');
    };
    const micBtn = document.querySelector('.mic-btn');
    if (micBtn) micBtn.classList.add('recording');
    this.recognition.start();
  },

  startGame(gameId) {
    this.playClick();
    this.currentGame = gameId;
    this.gameState = {
      score: 0,
      round: 0,
      maxRounds: 10,
      correct: 0,
      wrong: 0,

  // === GAME RENDER ===
  renderGame() {
    const g = this.gameList.find(x => x.id === this.currentGame);
    if (!g) return this.renderGames();
    
    const gs = this.gameState;
    if (gs.round >= gs.maxRounds) {
      return this.renderGameComplete(g);
    }

    // Generate game content berdasarkan type
    gs.round++;
    gs.data = this.generateGameData(this.currentGame);

    let content = '';
    switch(this.currentGame) {
      case 'tebakHuruf': content = this.gameTebakHuruf(); break;
      case 'balonHuruf': content = this.gameBalonHuruf(); break;
      case 'tangkapIkan': content = this.gameTangkapIkan(); break;
      case 'mobilBaca': content = this.gameMobilBaca(); break;
      case 'tebakGambar': content = this.gameTebakGambar(); break;
      case 'cocokkanKata': content = this.gameCocokkanKata(); break;
      case 'susunKata': content = this.gameSusunKata(); break;
      case 'warnaiHuruf': content = this.gameWarnaiHuruf(); break;
      case 'puzzleKata': content = this.gamePuzzleKata(); break;
      case 'hurufHilang': content = this.gameHurufHilang(); break;
      default: content = this.gameTebakHuruf();
    }

    const progressPercent = Math.round((gs.round / gs.maxRounds) * 100);

    return `
    <div class="game-area">
      <div style="font-family:'Fredoka One',cursive;font-size:14px;color:var(--text-medium);margin-bottom:8px;">
        ${g.icon} ${g.name} - Ronde ${gs.round}/${gs.maxRounds}
      </div>
      <div class="game-progress-bar">
        <div class="game-progress-fill" style="width:${progressPercent}%"></div>
      </div>
      <div class="game-score"><i class="fas fa-star"></i> ${gs.score} | <i class="fas fa-check-circle" style="color:var(--green);"></i> ${gs.correct} | <i class="fas fa-times-circle" style="color:var(--coral);"></i> ${gs.wrong}</div>
      ${content}
    </div>`;
  },

  renderGameComplete(g) {
    const gs = this.gameState;
    const percent = Math.round((gs.correct / gs.maxRounds) * 100);
    
    // Update progress
    this.progress.gamesPlayed[this.currentGame] = (this.progress.gamesPlayed[this.currentGame] || 0) + 1;
    this.progress.daily.gamesPlayed++;
    this.progress.stars += gs.score;
    if (percent >= 80) this.progress.medals++;
    this.progress.totalExercises += gs.maxRounds;
    this.saveProgress();

    // Tampilkan reward
    setTimeout(() => {
      this.showReward(percent >= 80 ? '🏆' : percent >= 50 ? '⭐' : '💪',
        percent >= 80 ? 'Luar Biasa!' : percent >= 50 ? 'Bagus Sekali!' : 'Ayo Coba Lagi!',
        `Skor: ${gs.score} | Benar: ${gs.correct}/${gs.maxRounds}`);
    }, 300);

    return `
    <div class="game-area" style="padding-top:40px;">
      <div style="font-size:80px;animation:rewardIconBounce 1s ease-in-out infinite;">${percent >= 80 ? '🏆' : percent >= 50 ? '⭐' : '💪'}</div>
      <div style="font-family:'Fredoka One',cursive;font-size:28px;color:var(--text-dark);margin-top:16px;">${percent >= 80 ? 'Luar Biasa!' : percent >= 50 ? 'Bagus Sekali!' : 'Ayo Coba Lagi!'}</div>
      <div style="font-size:16px;color:var(--text-medium);margin-top:8px;">Skor: ${gs.score} | Benar: ${gs.correct}/${gs.maxRounds} (${percent}%)</div>
      <div style="margin-top:24px;display:flex;gap:12px;justify-content:center;">
        <button class="home-btn btn-start" style="padding:12px 24px;font-size:14px;display:inline-flex;flex-direction:row;gap:6px;" onclick="App.startGame('${this.currentGame}')">
          <i class="fas fa-redo"></i> Main Lagi

        </button>
      </div>
    </div>`;
  },

  generateGameData(gameId) {
    switch(gameId) {
      case 'tebakHuruf': {
        const target = this.letters[Math.floor(Math.random()*26)];
        const options = [target.l];
        while(options.length < 4) {
          const r = this.letters[Math.floor(Math.random()*26)].l;
          if (!options.includes(r)) options.push(r);
        }
        return {target, options: this.shuffleArray(options), answer: target.l};
      }
      case 'balonHuruf': {
        const target = this.letters[Math.floor(Math.random()*26)];
        return {targetLetter: target.l, targetWord: target.w, targetEmoji: target.emoji};
      }
      case 'tangkapIkan': {
        const target = this.letters[Math.floor(Math.random()*26)];
        return {targetLetter: target.l};
      }
      case 'mobilBaca': {
        const word = this.words[Math.floor(Math.random()*this.words.length)];
        return {word: word.w, emoji: word.emoji, syllables: word.s};
      }
      case 'tebakGambar': {
        const word = this.words[Math.floor(Math.random()*this.words.length)];
        const options = [word.w];
        while(options.length < 4) {
          const r = this.words[Math.floor(Math.random()*this.words.length)].w;
          if (!options.includes(r)) options.push(r);
        }
        return {word, options: this.shuffleArray(options), answer: word.w};
      }
      case 'cocokkanKata': {
        const selected = this.shuffleArray(this.words).slice(0, 4);
        return {pairs: selected.map(w => ({word: w.w, emoji: w.emoji}))};
      }
      case 'susunKata': {
        const word = this.words[Math.floor(Math.random()*this.words.length)];
        const scrambled = this.shuffleArray(word.w.split(''));
        return {word: word.w, emoji: word.emoji, scrambled, placed: []};
      }
      case 'warnaiHuruf': {
        const target = this.letters[Math.floor(Math.random()*26)];
        const colors = ['#D32F2F','#E65100','#F57F17','#2E7D32','#1565C0','#6A1B9A'];
        return {letter: target.l, lower: target.l.toLowerCase(), word: target.w, emoji: target.emoji, 
                targetColor: colors[Math.floor(Math.random()*colors.length)],
                colorOptions: this.shuffleArray(colors).slice(0,4)};
      }
      case 'puzzleKata': {
        const word = this.words[Math.floor(Math.random()*this.words.length)];
        const missingIdx = Math.floor(Math.random()*word.w.length);
        const missingLetter = word.w[missingIdx];
        const display = word.w.split('');
        display[missingIdx] = '?';
        const options = [missingLetter];
        while(options.length < 4) {
          const r = this.letters[Math.floor(Math.random()*26)].l.toLowerCase();
          if (!options.includes(r)) options.push(r);
        }
        return {word: word.w, emoji: word.emoji, display: display.join(''), missingIdx, answer: missingLetter, options: this.shuffleArray(options)};
      }
      case 'hurufHilang': {
        const sentence = this.sentences[Math.floor(Math.random()*this.sentences.length)];
        const targetWord = sentence.words[Math.floor(Math.random()*sentence.words.length)];
        const missingIdx = Math.floor(Math.random()*targetWord.length);
        const missingLetter = targetWord[missingIdx];
        const display = targetWord.split('');
        display[missingIdx] = '_';
        const options = [missingLetter];
        while(options.length < 4) {
          const r = 'abcdefghijklmnopqrstuvwxyz'[Math.floor(Math.random()*26)];
          if (!options.includes(r)) options.push(r);
        }
        return {sentence, targetWord, display: display.join(''), missingIdx, answer: missingLetter, options: this.shuffleArray(options)};
      }
      default: return {};
    }
  },

  matchSelect(side, idx, value) {
    this.playClick();
    const ms = this.gameState.matchState;
    if (side === 'left') {
      // Clear previous left selection
      document.querySelectorAll('.match-item[id^="matchL"]').forEach(el => el.classList.remove('sel-left'));
      ms.selectedLeft = {idx, value};
      const el = document.getElementById(`matchL${idx}`);
      if (el) el.classList.add('sel-left');
    } else {
      document.querySelectorAll('.match-item[id^="matchR"]').forEach(el => el.classList.remove('sel-right'));
      ms.selectedRight = {idx, value};
      const el = document.getElementById(`matchR${idx}`);
      if (el) el.classList.add('sel-right');
    }
    // Check if both selected
    if (ms.selectedLeft && ms.selectedRight) {
      const leftWord = ms.selectedLeft.value;
      const rightEl = document.getElementById(`matchR${ms.selectedRight.idx}`);
      const rightWord = rightEl ? rightEl.dataset.word : '';
      if (leftWord === rightWord) {
        // Matched!
        this.playSuccess();
        const lel = document.getElementById(`matchL${ms.selectedLeft.idx}`);
        const rel = document.getElementById(`matchR${ms.selectedRight.idx}`);
        if (lel) lel.classList.add('matched');
        if (rel) rel.classList.add('matched');
        ms.matched.push(leftWord);
        this.gameState.score++;
        this.gameState.correct++;
        ms.selectedLeft = null;
        ms.selectedRight = null;
        // Check if all matched
        if (ms.matched.length >= 4) {
          setTimeout(() => this.navigate('game', this.currentGame), 800);
        }
      } else {
        this.playWrong();
        this.gameState.wrong++;
        setTimeout(() => {
          document.querySelectorAll('.match-item[id^="matchL"]').forEach(el => el.classList.remove('sel-left'));
          document.querySelectorAll('.match-item[id^="matchR"]').forEach(el => el.classList.remove('sel-right'));
          ms.selectedLeft = null;
          ms.selectedRight = null;
        }, 400);
      }
    }
  },

  susunClick(idx, letter) {
    this.playClick();
    const ss = this.gameState.susunState;
    const scrEl = document.getElementById(`scrambled${idx}`);
    if (scrEl.classList.contains('used')) return;
    scrEl.classList.add('used');
    
    ss.placed.push({letter, idx});
    const area = document.getElementById('arrangeArea');
    const letterEl = document.createElement('div');
    letterEl.className = 'arrange-letter placed';
    letterEl.textContent = letter;
    letterEl.onclick = () => {
      // Remove from placed
      ss.placed = ss.placed.filter(p => p.idx !== idx);
      scrEl.classList.remove('used');
      letterEl.remove();
    };
    area.appendChild(letterEl);

    // Check if complete
    const d = this.gameState.data;
    if (ss.placed.length === d.word.length) {
      const formed = ss.placed.map(p => p.letter).join('');
      if (formed === d.word) {
        this.playSuccess();
        this.gameState.score++;
        this.gameState.correct++;
        this.speak(d.word);
        setTimeout(() => this.navigate('game', this.currentGame), 1200);
      } else {
        this.playWrong();
        this.gameState.wrong++;
        // Reset
        setTimeout(() => {
          ss.placed = [];
          area.innerHTML = '';
          document.querySelectorAll('.scrambled-letter').forEach(el => el.classList.remove('used'));
        }, 400);
      }
    }
  },

  drawChart(canvasId) {
    const canvas = document.getElementById(canvasId);
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    const w = canvas.parentElement.clientWidth;
    const h = 160;
    canvas.width = w;
    canvas.height = h;

    const p = this.progress;
    // Simple bar chart
    const data = [
      {label:'Huruf', value: p.lettersLearned.length, max: 26, color: '#4D96FF'},
      {label:'Suku Kata', value: p.syllablesLearned.length, max: 55, color: '#66BB6A'},
      {label:'Kata', value: p.wordsRead.length, max: 15, color: '#FFD54F'},
      {label:'Kalimat', value: p.sentencesRead.length, max: 10, color: '#FFB347'},
      {label:'Cerita', value: p.storiesRead.length, max: 4, color: '#FF7F7F'}
    ];

    const barWidth = Math.min(50, (w - 60) / data.length - 10);
    const startX = 40;
    const maxBarH = h - 40;

    ctx.clearRect(0, 0, w, h);
    
    // Background
    ctx.fillStyle = '#f5f5f5';
    ctx.fillRect(0, 0, w, h);

    // Bars
    data.forEach((d, i) => {
      const x = startX + i * (barWidth + 10);
      const barH = Math.max(4, (d.value / d.max) * maxBarH);
      
      // Bar
      ctx.fillStyle = d.color;
      ctx.beginPath();
      ctx.roundRect(x, h - 20 - barH, barWidth, barH, 4);
      ctx.fill();
      
      // Label
      ctx.fillStyle = '#546E7A';
      ctx.font = '10px Nunito';
      ctx.textAlign = 'center';
      ctx.fillText(d.label, x + barWidth/2, h - 6);
      
      // Value
      ctx.fillStyle = '#37474F';
      ctx.font = 'bold 12px Nunito';
      ctx.fillText(d.value, x + barWidth/2, h - 24 - barH);
    });
  },

  spawnConfetti() {
    const container = document.getElementById('confettiContainer');
    const colors = ['#FF7F7F','#FFB347','#FFD54F','#66BB6A','#4D96FF','#CE93D8','#FF8A80','#A5D6A7'];
    let html = '';
    for (let i = 0; i < 60; i++) {
      const color = colors[Math.floor(Math.random()*colors.length)];
      const left = Math.random() * 100;
      const delay = Math.random() * 2;
      const size = 6 + Math.random() * 10;
      const shape = Math.random() > 0.5 ? 'border-radius:50%' : 'border-radius:2px';
      html += `<div class="confetti-piece" style="left:${left}%;width:${size}px;height:${size}px;background:${color};${shape};animation-delay:${delay}s;animation-duration:${2+Math.random()*2}s;"></div>`;
    }
    container.innerHTML = html;
    // Auto clear after 4s
    setTimeout(() => { container.innerHTML = ''; }, 4000);
  },

  shuffleArray(arr) {
    const a = [...arr];
    for (let i = a.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [a[i], a[j]] = [a[j], a[i]];
    }
    return a;
  },

// Polyfill roundRect untuk canvas
if (!CanvasRenderingContext2D.prototype.roundRect) {
  CanvasRenderingContext2D.prototype.roundRect = function(x, y, w, h, r) {
    r = Math.min(r, w/2, h/2);
    this.moveTo(x+r, y);
    this.arcTo(x+w, y, x+w, y+h, r);
    this.arcTo(x+w, y+h, x, y+h, r);
    this.arcTo(x, y+h, x, y, r);
    this.arcTo(x, y, x+w, y, r);
    this.closePath();
  };
}
